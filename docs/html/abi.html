Native ABI v1

Контракт между Core и DLL: include/operator.h, стабильный, только C17.

1. Точка входа

DLL экспортирует ровно один символ:

const VxOpInfo *vxop_open(const VxOpApi *api);
typedef struct VxOpFuncInfo { const char *name; int arity; VxOpFunc f; } ...;
typedef struct VxOpInfo {
    int abi_version;      /* VXOP_ABI_VERSION */
    const char *name;     /* == manifest */
    const char *version;  /* == manifest */
    int nfuncs;
    const VxOpFuncInfo *funcs;
} VxOpInfo;

Загрузчик проверяет: экспорт есть, abi_version совпал, имя+версия совпали с манифестом. Иначе — громкий отказ (Operator ABI mismatch, Failed to load Operator).

2. Владение — весь контракт в пяти строках

  • Аргументы принадлежат Core. Не освобождать их.
  • *out — всегда СВЕЖЕЕ: конструкторы api или clone. Никогда не отдавать сам указатель args (он занимает память стека VM — это роняло кучу, поймано тестом).
  • retain — только значениям, переживающим вызов; освободить до выгрузки.
  • func возвращает 0; ненулевое поднимает fail (лечится через ??).
  • Значения трогать только через api — напрямую память VM недоступна в принципе (тип непрозрачный).

4. Обратный призыв (reentrancy)

Оператор может дёрнуть Vexel-код посреди своего вызова — так run в VexGUI каждый раунд призывает step. VM выполняет вложенный вызов на тех же стеках с явной глубиной (run_until), предел вложенности — 256 кадров. Ошибка шага печатается в stderr с именем инструмента, вызов возвращает ненулевое.

5. Минимальный оператор целиком

#include "operator.h"

static int op_ping(const VxOpApi *api, VxOpVal **args, int argc,
                   VxOpVal **out) {
    (void)argc;
    *out = api->clone(args[0]);
    return *out ? 0 : 1;
}

static const VxOpFuncInfo funcs[] = { { "ping", 1, op_ping } };

#ifdef _WIN32
#define VXOP_EXPORT __declspec(dllexport)
#else
#define VXOP_EXPORT
#endif

VXOP_EXPORT const VxOpInfo *vxop_open(const VxOpApi *api) {
    static VxOpInfo info;
    (void)api;
    info.abi_version = VXOP_ABI_VERSION;
    info.name = "MyOp";
    info.version = "0.1.0";
    info.nfuncs = 1;
    info.funcs = funcs;
    return &info;
}
